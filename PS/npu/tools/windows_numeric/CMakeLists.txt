cmake_minimum_required(VERSION 3.18)
project(pha_numeric_windows LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Icraft-XRT CONFIG REQUIRED)
find_package(icraft-hostbackend CONFIG REQUIRED)
add_executable(validate_numeric_windows ../validate_numeric.cpp)
target_include_directories(validate_numeric_windows PRIVATE ../../thirdparty)
target_link_libraries(validate_numeric_windows PRIVATE Icraft::XRT Icraft::HostBackend)
if(MSVC)
    target_compile_options(validate_numeric_windows PRIVATE /utf-8 /Zc:__cplusplus)
endif()
