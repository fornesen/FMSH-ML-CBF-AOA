cmake_minimum_required(VERSION 3.16)
project(icraft_npu_yolo LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
option(PHA_NPU_HOST_ONLY "Build preprocessing/manifest validation without Icraft runtime" OFF)
option(PHA_NPU_BUILD_DECODER "Build the host numerical calibration decoder" OFF)
add_executable(icraft_npu_yolo src/main.cpp src/io.cpp src/preprocess.cpp src/icraft_runtime.cpp)
target_include_directories(icraft_npu_yolo PRIVATE include thirdparty)
target_link_libraries(icraft_npu_yolo PRIVATE pthread rt)
if(PHA_NPU_HOST_ONLY)
    target_compile_definitions(icraft_npu_yolo PRIVATE PHA_NPU_HOST_ONLY)
    add_executable(ipc_service_tests tests/ipc_service_tests.cpp)
    target_include_directories(ipc_service_tests PRIVATE include thirdparty)
    target_link_libraries(ipc_service_tests PRIVATE pthread rt)
    add_executable(classification_contract_tests tests/classification_contract.cpp src/preprocess.cpp src/io.cpp)
    target_include_directories(classification_contract_tests PRIVATE include thirdparty)
    target_compile_options(classification_contract_tests PRIVATE -Wall -Wextra -Werror)
    add_executable(amplitude_preprocessing_tests tests/amplitude_preprocessing.cpp src/preprocess.cpp src/io.cpp)
    target_include_directories(amplitude_preprocessing_tests PRIVATE include thirdparty)
    target_compile_options(amplitude_preprocessing_tests PRIVATE -Wall -Wextra -Werror)
    add_executable(model_identity_tests tests/model_identity.cpp src/io.cpp)
    target_include_directories(model_identity_tests PRIVATE include thirdparty)
    target_compile_options(model_identity_tests PRIVATE -Wall -Wextra -Werror)
    enable_testing()
    add_test(NAME classification_contract COMMAND classification_contract_tests)
    add_test(NAME amplitude_preprocessing COMMAND amplitude_preprocessing_tests)
    add_test(NAME model_identity COMMAND model_identity_tests)
    add_executable(dump_amplitude_features tests/dump_amplitude_features.cpp src/preprocess.cpp src/io.cpp)
    target_include_directories(dump_amplitude_features PRIVATE include thirdparty)
    target_compile_options(dump_amplitude_features PRIVATE -Wall -Wextra -Werror)
else()
    find_package(Icraft-XRT CONFIG REQUIRED)
    find_package(icraft-hostbackend CONFIG REQUIRED)
    find_package(icraft-zg330backend CONFIG REQUIRED)
    target_link_libraries(icraft_npu_yolo PRIVATE Icraft::XRT Icraft::HostBackend Icraft::ZG330Backend)
    if(PHA_NPU_BUILD_DECODER)
        add_library(pha_calibration_decoder SHARED tools/calibration_decoder.cpp)
        target_include_directories(pha_calibration_decoder PRIVATE thirdparty)
        target_link_libraries(pha_calibration_decoder PRIVATE Icraft::XRT)
        add_executable(inspect_compiled tools/inspect_compiled.cpp)
        target_include_directories(inspect_compiled PRIVATE thirdparty)
        target_link_libraries(inspect_compiled PRIVATE Icraft::XRT Icraft::HostBackend Icraft::ZG330Backend)
        add_executable(validate_numeric tools/validate_numeric.cpp)
        target_include_directories(validate_numeric PRIVATE thirdparty)
        target_link_libraries(validate_numeric PRIVATE Icraft::XRT Icraft::HostBackend)
    endif()
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(icraft_npu_yolo PRIVATE -Wall -Wextra -Werror)
endif()
